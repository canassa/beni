import Html exposing (Html)



pub type alias Model =
    name6502 : String
    email6502 : String
    submitted6502 : Bool
    attempts6502 : Int

pub type Msg
    = Name String
    | Email String
    | Submit
    | Reset

pub init : ⊤ → Model
init ⊤ = { name6502 = "", email6502 = "", submitted6502 = False, attempts6502 = 0 }

pub update : Msg, Model → Model
update msg model =
    case msg of
        Name text →
            if String.contains text "#" then model else { model | name6502 = text, submitted6502 = False }

        Email text →
            { model | email6502 = text, submitted6502 = False }

        Submit →
            { model | submitted6502 = True, attempts6502 = model.attempts6502 + 1 }

        Reset →
            { model | name6502 = "", email6502 = "", submitted6502 = False }

pub view : Model → Html Msg
view model =
    <section id="s0" data-shape="form"><h2>form 17/0 People</h2><label>Name<input id="s0-name" value={model.name6502} onInput={Name} /></label><label>Email<input id="s0-email" value={model.email6502} onInput={Email} /></label><p id="s0-validation">{if String.length model.name6502 < 4 then "Name too short" else if String.contains model.email6502 "@" then "Ready" else "Email needs @"}</p><button id="s0-submit" onClick={Submit}>Submit</button><button id="s0-reset" onClick={Reset}>Reset</button><p hidden={not model.submitted6502}>Saved {model.name6502} / {model.attempts6502}</p></section>
