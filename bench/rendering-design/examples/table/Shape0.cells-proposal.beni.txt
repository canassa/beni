-- Proposed surface, not accepted by today's compiler.
-- Action is a typed, deferred DOM callback, not an ordinary eager call.
-- An action commits its transaction atomically; render runs after commit.
-- Model snapshots do not escape the owned component.
import Html exposing (Html, Action)

component Instance
    type alias Row =
        id : Int
        title : String
        score : Int
        done : Bool

    type alias Model =
        rows2266 : List Row
        next2266 : Int
        selected2266 : Int
        descending2266 : Bool


    init : ⊤ → Model
    init ⊤ = { rows2266 = List.map (List.range 1 20) λi → { id = i, title = "Item " ++ String.fromInt i, score = Int.mod (i * 6) 17, done = False }, next2266 = 21, selected2266 = 0, descending2266 = False }

    state model : Model = init ⊤

    action Add =
        transaction
            let old = model
            model ←
                { old | rows2266 = [ …old.rows2266, { id = old.next2266, title = "table 17/0 People " ++ String.fromInt old.next2266, score = 6, done = False } ], next2266 = old.next2266 + 1 }

    action Remove (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows2266 = List.filter old.rows2266 λrow → row.id ≠ wanted, selected2266 = 0 }

    action Edit (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows2266 = List.map old.rows2266 λrow → if row.id == wanted then { row | title = row.title ++ "!", score = row.score + 6, done = not row.done } else row }

    action Select (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | selected2266 = wanted }

    action Reverse =
        transaction
            let old = model
            model ←
                { old | rows2266 = List.reverse old.rows2266 }

    action Sort =
        transaction
            let old = model
            model ←
                { old | descending2266 = not old.descending2266 }

    view : Html Action
    view =
        <section id="s0" data-shape="table"><h2>table 17/0 People</h2><div class="level-0"><button id="s0-add" onClick={Add}>Add</button><button id="s0-reverse" onClick={Reverse}>Reverse</button><p>{List.length model.rows2266} records</p><table><thead><tr><th><button id="s0-sort" onClick={Sort}>Sort</button></th><th>Score</th><th>Actions</th></tr></thead><tbody><For each={if model.descending2266 then List.reverse (List.sortBy model.rows2266 .score) else List.sortBy model.rows2266 .score} keyed={.id}>{λrow → <tr data-row={String.fromInt row.id} class={if model.selected2266 == row.id then "selected" else ""}><td><button onClick={Select row.id}>{row.title}</button></td><td>{row.score}</td><td><button onClick={Edit row.id}>Edit</button><button onClick={Remove row.id}>Remove</button></td></tr>}</For></tbody></table></div></section>
