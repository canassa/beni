-- Proposed surface, not accepted by today's compiler.
-- Action is a typed, deferred DOM callback, not an ordinary eager call.
-- An action commits its transaction atomically; render runs after commit.
-- Model snapshots do not escape the owned component.
import Html exposing (Html, Action)

component Instance
    type alias Row =
        id : Int
        title : String
        score : Int
        done : Bool

    type alias Model =
        rows2595 : List Row
        next2595 : Int
        selected2595 : Int
        query2595 : String
        onlyDone2595 : Bool


    init : ⊤ → Model
    init ⊤ = { rows2595 = List.map (List.range 1 20) λi → { id = i, title = "Item " ++ String.fromInt i, score = Int.mod (i * 1) 17, done = False }, next2595 = 21, selected2595 = 0, query2595 = "", onlyDone2595 = False }

    state model : Model = init ⊤

    action Add =
        transaction
            let old = model
            model ←
                { old | rows2595 = [ …old.rows2595, { id = old.next2595, title = "filter 17/0 Activity " ++ String.fromInt old.next2595, score = 1, done = False } ], next2595 = old.next2595 + 1 }

    action Remove (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows2595 = List.filter old.rows2595 λrow → row.id ≠ wanted }

    action Edit (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows2595 = List.map old.rows2595 λrow → if row.id == wanted then { row | title = row.title ++ "!", score = row.score + 1, done = not row.done } else row }

    action Select (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | selected2595 = wanted }

    action Reverse =
        transaction
            let old = model
            model ←
                { old | rows2595 = List.reverse old.rows2595 }

    action Query (text : String) =
        transaction
            let old = model
            model ←
                { old | query2595 = text }

    action OnlyDone =
        transaction
            let old = model
            model ←
                { old | onlyDone2595 = not old.onlyDone2595 }

    view : Html Action
    view =
        <section id="s0" data-shape="filter"><h2>filter 17/0 Activity</h2><div class="level-0"><button id="s0-add" onClick={Add}>Add</button><button id="s0-reverse" onClick={Reverse}>Reverse</button><input id="s0-query" value={model.query2595} onInput={Query} /><button id="s0-onlydone" onClick={OnlyDone}>OnlyDone</button><p>{List.length model.rows2595} records</p><ul><For each={List.filter model.rows2595 λrow → String.contains row.title model.query2595 && (not model.onlyDone2595 || row.done)} keyed={.id}>{λrow → <li data-row={String.fromInt row.id} class={if model.selected2595 == row.id then "selected" else ""}><button onClick={Select row.id}>{row.title}</button><output>{row.score}</output><button onClick={Edit row.id}>Edit</button><button onClick={Remove row.id}>Remove</button></li>}</For></ul></div></section>
