-- Proposed surface, not accepted by today's compiler.
-- Action is a typed, deferred DOM callback, not an ordinary eager call.
-- An action commits its transaction atomically; render runs after commit.
-- Model snapshots do not escape the owned component.
import Html exposing (Html, Action)

component Instance
    type alias Row =
        id : Int
        title : String
        score : Int
        done : Bool

    type alias Model =
        rows1346 : List Row
        next1346 : Int
        selected1346 : Int


    init : ⊤ → Model
    init ⊤ = { rows1346 = List.map (List.range 1 20) λi → { id = i, title = "Item " ++ String.fromInt i, score = Int.mod (i * 5) 17, done = False }, next1346 = 21, selected1346 = 0 }

    state model : Model = init ⊤

    action Add =
        transaction
            let old = model
            model ←
                { old | rows1346 = [ …old.rows1346, { id = old.next1346, title = "list 17/0 People " ++ String.fromInt old.next1346, score = 5, done = False } ], next1346 = old.next1346 + 1 }

    action Remove (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows1346 = List.filter old.rows1346 λrow → row.id ≠ wanted, selected1346 = 0 }

    action Edit (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | rows1346 = List.map old.rows1346 λrow → if row.id == wanted then { row | title = row.title ++ "!", score = row.score + 5, done = not row.done } else row }

    action Select (wanted : Int) =
        transaction
            let old = model
            model ←
                { old | selected1346 = wanted }

    action Reverse =
        transaction
            let old = model
            model ←
                { old | rows1346 = List.reverse old.rows1346 }

    view : Html Action
    view =
        <section id="s0" data-shape="list"><h2>list 17/0 People</h2><button id="s0-add" onClick={Add}>Add</button><button id="s0-reverse" onClick={Reverse}>Reverse</button><p>{List.length model.rows1346} records</p><ul><For each={model.rows1346} keyed={.id}>{λrow → <li data-row={String.fromInt row.id} class={if model.selected1346 == row.id then "selected" else ""}><button onClick={Select row.id}>{row.title}</button><output>{row.score}</output><button onClick={Edit row.id}>Edit</button><button onClick={Remove row.id}>Remove</button></li>}</For></ul></section>
